python -m torch.distributed.launch --nproc_per_node=3 --master_port 12345 train.py \
     --dataset TN3K \
     --root_path /home/ub/pql/Swin-Unet-main/data/TN3K \
     --list_dir /home/ub/pql/Swin-Unet-main/lists/TN3K \
     --n_class 2 \
     --output_dir ./output_tn3k \
     --img_size 384 \
     --batch_size 2 \
     --max_epochs 100 \
     --base_lr 0.01 \
     --cfg configs/swin_base_patch4_window12_384_22k.yaml
