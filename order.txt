git status
这会列出哪些文件被修改了，哪些是新增加的。这能帮你检查是否有不小心改错的文件。

# 第一步：把所有修改过的文件添加到暂存区
git add .

# 第二步：提交更改并写下说明（建议写清楚你改了什么）
git commit -m "优化了特征融合模块的代码并调整了超参数"

# 第三步：推送到 GitHub 远程仓库
git push origin main

  --dataset TN3K \
  --root_path /home/ub/pql/Swin-Unet-main/data/TN3K \
  --list_dir /home/ub/pql/Swin-Unet-main/lists/TN3K \
  --cfg configs/swin_base_patch4_window12_384_22k.yaml \
  --img_size 384 \
  --n_class 2 \
  --output_dir ./output_tn3k \
  --batch_size 4 \
  --max_epochs 150 \
  --base_lr 1e-4 \
  --optimizer adamw \
  --edge_lambda 0.1 \
  --boundary_width 3 \
  --edge_focal_gamma 2.0 \
  --edge_warmup_ratio 0.1 \
  --edge_decay_start_ratio 0.6 \
  --edge_lambda_min_factor 0.5 \
  --num_workers 4
